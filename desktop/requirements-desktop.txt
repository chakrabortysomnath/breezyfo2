# Breezy F&O — Desktop dependencies
#
# Everything needed to run the backend + frontend + native window locally on a
# laptop (Python assumed present). Install into a virtualenv via desktop/setup.ps1.
#
# Mirrors backend/requirements.txt and frontend/requirements.txt, plus pywebview
# for the native window. Versions are LOWER-BOUND ranges (not the hard pins used
# in the service requirements) so pip installs wheels that match the local
# interpreter — the hard pins (pydantic 2.7.1, bcrypt 3.2.2, sqlalchemy 2.0.30)
# have no wheels for Python 3.12/3.13 and would fail to build there.

# ---- Backend (FastAPI + Breeze + SQLite + JWT auth) ----

pydantic[email]>=2.7
pydantic-settings>=2.2
python-dotenv>=1.0
httpx>=0.27

PySocks>=1.7.1
sqlalchemy>=2.0.30
passlib[bcrypt]>=1.7.4
bcrypt>=4.0             # modern wheels incl. Python 3.13; passlib works with 4.x
python-jose[cryptography]>=3.3.0

# ---- Frontend (Streamlit UI) ----
streamlit>=1.40,<2.0
requests>=2.32
pandas>=2.2.3,<3        # app targets pandas 2.x; avoid 3.0 breaking changes

# ---- Desktop shell ----
pywebview               # native window (WebView2 on Windows 11)
